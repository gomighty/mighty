## Two files, end to end

Wire a backend route to an Astro page, hand it typed props, and let Astro render. That is the whole integration.

</div>

<CodeCards>
  <CodeCard file="src/server.ts">
    <HonoLogo slot="logo" />

    ```ts frame="none"
    import { Hono } from "hono";
    import { mighty } from "@gomighty/hono";

    const app = new Hono();
    app.use(mighty());

    app.get("/posts", async (c) => {
      const posts = await db.posts.all();
      // render an Astro page, pass typed props
      return c.render({ component: "posts.index", props: { posts } });
    });
    ```

  </CodeCard>
  <CodeCard file="astro/pages/posts/index.astro">
    <AstroLogo slot="logo" />

    ```astro frame="none"
    ---
    interface Props {
      posts: Post[];
    }

    const { posts } = Astro.props;
    ---

    <ul class="feed">
      {posts.map((post) => (
        <li>
          <h2>{post.title}</h2>
          <p>{post.excerpt}</p>
        </li>
      ))}
    </ul>
    ```

  </CodeCard>
</CodeCards>
